1️⃣  GET BOT TOKEN
   https://discord.com/developers/applications
   → New Application → Bot → Reset Token

2️⃣  RUN INSTALLER
   bash install.sh

3️⃣  ADD TOKEN
   nano .env
   → Paste: DISCORD_TOKEN=your_token_here

4️⃣  START BOT
   python3 bot.py

5️⃣  TEST IN DISCORD
   $play test

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PLAYBACK
  $play <song>      Play music
  $skip             Next song
  $back             Previous song
  $pause            Pause

  $nowplaying       Current song
  $shuffle          Randomize

EFFECTS
  $volume 150       Set volume
  $nightcore        Speed up
  $bassboost        More bass

VOICE
  $voice join       Join your channel
  $voice stay       Stay 24/7

📝 All commands work with / too!

🐛 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ Bot not responding?
   → Enable Message Content Intent
   → Check bot has permissions

❌ Music not playing?
   → Install FFmpeg: sudo apt install ffmpeg
   → Join voice channel first

❌ Slash commands missing?
   → Wait 1 hour for Discord sync
   → Re-invite with applications.commands scope

❌ Lyrics error?
   → Add GENIUS_ACCESS_TOKEN to .env

❌ Bot crashes?
   → Check: tail -f bot.log
   → Verify Python 3.10+
